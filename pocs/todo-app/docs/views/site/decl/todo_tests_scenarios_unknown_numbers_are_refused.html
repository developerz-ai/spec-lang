@given(board.three_open_one_done)
test unknown_numbers_are_refused(no: TaskNo) = {
  if no <= 4 { return; }
  board.done(no) -> fail NoTask;
  board.reopen(no) -> fail NoTask;
  board.rename(no, title: "x") -> fail NoTask;
  board.remove(no) -> fail NoTask;
  assert out.text == "";
}
