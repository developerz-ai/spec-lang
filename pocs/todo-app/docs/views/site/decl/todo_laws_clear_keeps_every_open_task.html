law clear_keeps_every_open_task(t: Trace[Change]) =
  t.steps.all(s => s.event.is(board.clear()) implies
    s.after.tasks.values.all(it.status == Open)
    and s.after.tasks.size == s.before.tasks.values.filter(it.status == Open).size);
