@why("a title fits on one line of a terminal list")
@cases("milk" -> true, "" -> false, "   " -> false, "a\tb" -> false, "\u{3000}" -> false, "a\u{2028}b" -> false, "a\u{0085}b" -> false)
pub fn is_title(t: Text) -> Bool =
  t.trim().size >= 1 and t.size <= 200 and t.bytes.all(it >= 32 and it != 127)
  and not t.contains("\u{0085}") and not t.contains("\u{2028}") and not t.contains("\u{2029}");
