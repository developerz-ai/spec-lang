// todo.laws: what must hold after any sequence of commands, in any order.

use todo.board;
use todo.model.TaskNo;
use todo.model.Status;

type Change = board.add | board.done | board.reopen | board.rename | board.remove | board.clear;

@why("a number is never reused")
law numbers_below_next(t: Trace[Change]) =
  t.states.all(w => w.tasks.keys.all(it < w.next_no));

@why("only add moves the counter, and only up")
law next_only_grows(t: Trace[board.add]) =
  t.steps.all(it.after.next_no >= it.before.next_no);

@why("a command that names a task touches no other task")
law others_untouched(t: Trace[board.done | board.reopen | board.rename | board.remove], k: TaskNo) =
  t.steps.all(s => s.event.no != k implies s.after.tasks.get(k) == s.before.tasks.get(k));

law clear_keeps_every_open_task(t: Trace[Change]) =
  t.steps.all(s => s.event.is(board.clear()) implies
    s.after.tasks.values.all(it.status == Open)
    and s.after.tasks.size == s.before.tasks.values.filter(it.status == Open).size);

law reads_change_nothing(t: Trace[board.list | board.stats]) =
  t.steps.all(it.after == it.before);

@why("a refused command leaves the board as it was")
law refusal_changes_nothing(t: Trace[Change]) =
  t.steps.all(s => s.reply.is(Err(_)) implies s.after == s.before);

@why("a task is stored under its own number")
law stored_under_its_number(t: Trace[Change]) =
  t.states.all(w => w.tasks.keys.all(k => w.tasks[k].no == k));
