# case: TODO_STORE moves the store
& TODO_STORE=lists/home.json
$ ["add", "a"]
! error: store lists/home.json: unwritable
? 3
$ ["list"]
! error: store lists/home.json: unwritable
? 3
& TODO_STORE=home.json
$ ["add", "a"]
> added 1
= {"version":1,"next_no":2,"tasks":[{"no":1,"title":"a","status":"Open","priority":"Normal","tags":[]}]}
