# The rules of todo.board, end to end.

# case: empty board
$ ["list"]
> nothing to do
$ ["list", "--all"]
> nothing to do
$ ["stats"]
> 0 open, 0 done
= absent

# case: add numbers from one and writes the canonical store
$ ["add", "milk"]
> added 1
$ ["add", "bread", "--tags", "shop", "--tags", "home", "--priority", "high"]
> added 2
$ ["list"]
> 2 [ ] bread !high #home #shop
> 1 [ ] milk
= {"version":1,"next_no":3,"tasks":[{"no":1,"title":"milk","status":"Open","priority":"Normal","tags":[]},{"no":2,"title":"bread","status":"Open","priority":"High","tags":["home","shop"]}]}

# case: numbers are never reused
$ ["add", "a"]
> added 1
$ ["add", "b"]
> added 2
$ ["remove", "2"]
> removed 2
$ ["add", "c"]
> added 3

# case: done and reopen move once
$ ["add", "a"]
> added 1
$ ["done", "1"]
> 1 [x] a
$ ["done", "1"]
! error: AlreadyDone
? 1
$ ["reopen", "1"]
> 1 [ ] a
$ ["reopen", "1"]
! error: NotDone
? 1
